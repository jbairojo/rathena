	for(.@a = 0; .@a < .@len; .@a++) {

-	script	HourlyPoints	-1,{
OnInit:
	.Kpoints = 10;
	.OnMinute = 60;
	.npc_name$ = strnpcinfo(3);
	end;

OnPointGet:
	#HourlyPoints++;
	switch ( #HourlyPoints ) {
		default:
			break;
		case 60:
			#HourlyPoints = -1;
			#KAFRAPOINTS = #KAFRAPOINTS + .Kpoints;
			dispbottom "You received "+.Kpoints+" Kafra points. Total of "+#KAFRAPOINTS+" points.",0x00ff00;
			deltimer .npc_name$+"::OnPointGet";
			break;
	}

OnPCLoginEvent:
	if ( #HourlyPoints < .OnMinute ) {
		addtimer (1000*60),.npc_name$+"::OnPointGet";
	}
	end;
}
