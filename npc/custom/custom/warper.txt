-	script	Warper#quest	-1,{
function Go;
	.@itemid = 6735;
	cutin "kafra_do01", 2;
	mes "[Kafra Warper]";
	mes "Welcome to Kafra Services.";
	mes "Our staff is always working";
	mes "to surpass your expactations";
	mes "for quality service.";
	mes "So how may I assist you today?";
	next;

menu
	"Last Warp [^777777"+lastpick$+"^000000]",-,
	"Non-Kafra Towns",L_town,
	"Dungeons",L_dungeon,
	"Cancel",L_end;

	mes "[Kafra Warper]";
	if (lastwarp$ == "") {
		mes "You haven't unlocked";
		mes "any warp points yet.";
		close3;
	} else if (!countitem(.@itemid)) {
		mes "I'm sorry, but you do not have a";
		mes callfunc("itemlink",.@itemid)+"  for this service.";
		mes "Please check back again.";
		close3;
	}
	mes "Have a safe journey.";
	delitem .@itemid,1;
	warp lastwarp$,lastwarpx,lastwarpy;
	end;

function Go {
	if (!countitem(.@itemid)) {
		mes "[Kafra Warper]";
		mes "I'm sorry, but you do not have a";
		mes callfunc("itemlink",.@itemid)+" for this service.";
		mes "Please check back again.";
		close3;
	}
	delitem .@itemid,1;
	set lastwarp$, getarg(0);
	set lastwarpx, getarg(1,0);
	set lastwarpy, getarg(2,0);
	set lastpick$, getarg(3);
	warp getarg(0),getarg(1,0),getarg(2,0);
	end;
}

// --------------------------------------------------
//Towns
// --------------------------------------------------

L_town:
	mes "[Kafra Warper]";
	mes "Please choose your destination.";
	next;
	callfunc "QWS_Tarray";
	freeloop(1);
	set .@menu$,"";
	for(set .@i,0; .@i<35; set .@i,.@i+1)	// Expected maximum is 35
		set .@menu$,.@menu$+@Tmenulist$[.@i]+":";
	freeloop(0);
	set @TWMenu,select(.@menu$);

	if (@Tmenuref[@TWMenu-1] == 57005) goto L_end; // 57005='dead' in hex

	set .@x, @pTXcoords[@Tmenuref[@TWMenu-1]];
	set .@y, @pTYcoords[@Tmenuref[@TWMenu-1]];
	Go(@pTmap$[@Tmenuref[@TWMenu-1]],.@x,.@y,@pTmenuitems$[@Tmenuref[@TWMenu-1]]);
	end;

// --------------------------------------------------
//Dungeons
// --------------------------------------------------

L_dungeon:
	mes "[Kafra Warper]";
	mes "Please choose your destination.";
	next;
	callfunc "QWS_Darray";
	freeloop(1);
	set .@menu$,"";
	for(set .@i,0; .@i<35; set .@i,.@i+1)	// Expected maximum is 35
		set .@menu$,.@menu$+@Dmenulist$[.@i]+":";
	freeloop(0);
	set @DWMenu,select(.@menu$);

	if (@Dmenuref[@DWMenu-1] == 57005) goto L_end; // 57005='dead' in hex

	set .@x, @DXcoords[@Dmenuref[@DWMenu-1]];
	set .@y, @DYcoords[@Dmenuref[@DWMenu-1]];
	Go(@DGat$[@Dmenuref[@DWMenu-1]],.@x,.@y,@pDmenuitems$[@Dmenuref[@DWMenu-1]]);

L_end:
	mes "[Kafra Warper]";
	mes "The Kafra Corporation is always";
	mes "always striving to provide the best";
	mes "services for our customers.";
	close3;
}

//----------------Make Town Menu Function

function	QWS_Make_Town_Menu	{
	set @temptownmenubin,1;
	if (getarg(0) == 0) goto menu_item;
	set @templooptownmenu,0;
	do {
		set @temptownmenubin, @temptownmenubin * 2;
		set @templooptownmenu, @templooptownmenu + 1;
		}while (getarg(0) > @templooptownmenu);
// check marker and make menu item
menu_item:
	if ((@temptownmenubin & QWS_T_GO) != @temptownmenubin) {
		set @Ti,@Ti+1;
		return;
	}
	if (@pTprice[@Ti] == 0)&&($QW_MapUserShow == 0) set @Tmenulist$[@Tj], @pTmenuitems$[@Ti];
	set @Tmenuref[@Tj],@Ti;
	set @Tj,@Tj+1;
	set @Ti,@Ti+1;
	return;
}
}

function	QWS_Make_Dungeon_Menu	{
	set @tempdungeonmenubin,1;
	if (getarg(0) == 0) goto menu_item;
	set @temploopdungeonmenu,0;
	do {
		set @tempdungeonmenubin, @tempdungeonmenubin * 2;
		set @temploopdungeonmenu, @temploopdungeonmenu + 1;
		}while (getarg(0) > @temploopdungeonmenu);
// check marker and make menu item
menu_item:
	if ((@tempdungeonmenubin & QWS_D_GO) != @tempdungeonmenubin) {
		set @Di,@Di+1;
		return;
	}
	if (@pDprice[@Di] == 0)&&($QW_MapUserShow == 0) set @Dmenulist$[@Dj], @pDmenuitems$[@Di];
	set @Dmenuref[@Dj],@Di;
	set @Dj,@Dj+1;
	set @Di,@Di+1;
	return;
}
}

//================Dungeon=Warpras=That=Only=Unlock=================

function	script	QWS_Dungeon_Warp	{
	function QWS_D_getbin;
	function QWS_D_setbin;

	if ((QWS_D_getbin(getarg(0)) == 0) && (getarg(1) != "")) {
		specialeffect2 EF_FLOWERLEAF;
		message strcharinfo(0),getarg(1)+" has been unlocked.";
		QWS_D_setbin(getarg(0));
		return;
	} else if (QWS_D_getbin(getarg(0)) == 1) {
		end;
	} else
		debugmes "QWS_Dungeon_Warp error, improper syntax?";
	return;


function	QWS_D_setbin	{
	set @tempsetbindata,1;
	if (getarg(0) == 0) goto binset;
	set @temploopsetbin,0;
	do {
		set @tempsetbindata, @tempsetbindata * 2;
		set @temploopsetbin, @temploopsetbin + 1;
	} while (getarg(0) > @temploopsetbin);
binset:
	set QWS_D_GO,(QWS_D_GO | @tempsetbindata);
	return;
}

function	QWS_D_getbin	{
	set @tempgetbindata,1;
	if (getarg(0) == 0) goto binget;
	set @temploopgetbin,0;
	do {
		set @tempgetbindata, @tempgetbindata * 2;
		set @temploopgetbin, @temploopgetbin + 1;
	} while (getarg(0) > @temploopgetbin);
binget:
	if ((@tempgetbindata & QWS_D_GO) == @tempgetbindata) return 1;
	return 0;
}
}
