//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prt_in01,252,166,5	script	Hypnotist#main	893,{
	setarray .@Reset,	0,	0,	0;
	 if (BaseLevel > 69) {
		setarray .@Reset,	2000000,	5000000,	200000,	6734,	25,	5;
	}
	mes "[Hypnotist]";
	mes "I'm a member of the Hypnotist";
	mes "Academy sent here to provide";
	mes "Stat and Skill Reset services.";
	next;
	mes "[Hypnotist]";
	if (!.@Reset) {
		mes "Characters below ^0000FFBase Lv. 70^000000. are offered free Stat and Skill Reset.";
		mes "Please select an option.";
		next;
		set .@i,(select("Reset Stats:Reset Skills:Cancel"));
		if(.@i > 2) close;
		if(.@i&1) ResetStatus;
		else if(.@i&2) {
			sc_end_class;
			ResetSkill;
		}
		mes "[Hypnotist]";
		mes "All done!";
		close;
	}
	mes "Stat Reset Limit: ^0000FF["+reset_limit+" / "+.@Reset[5]+"]^000000";
	mes "Stat Reset Rate: ^0000FF"+callfunc("F_InsertComma",.@Reset[2])+" z^000000";
	mes "Skill Reset Rate: ^0000FF"+callfunc("F_InsertComma",.@Reset[0])+" z^000000";
	mes "Full Stat Reset:";
	mes "     > ^0000FF"+callfunc("F_InsertComma",.@Reset[1])+" z^000000";
	mes "     > ^0000FF"+callfunc("F_InsertComma",.@Reset[4])+" "+getitemname(.@Reset[3])+"^000000";
	next;
	switch(select("Reset 1 Stat Point:Reset ALL Stats:Reset Skills:Cancel")) {
	case 1:
		mes "[Hypnotist]";
		mes "Please select the Stat Point you want to reset.";
		next;
		.@s = select("-1 Str","-1 Agi","-1 Vit","-1 Int","-1 Dex","-1 Luk") -1;
		mes "[Hypnotist]";
		for (.@i = 0; .@i < 10; ++.@i) {
			if (readparam(.stat[.@s]) == 1)
				break;
			if (Zeny < .@Reset[2]) {
				mes "Sorry, but you don't have enough zeny.";	
				close;
			}
			set Zeny, Zeny-.@Reset[2];
			statusup2 .stat[.@s], -1;
			StatusPoint += needed_status_point(.stat[.@s], 1);
			mes "All done!";
			close;
		}
		mes "This Stat Point can't be reset any further.";
		close;
	case 2:
		mes "[Hypnotist]";
		if(.@Reset[5] && reset_limit > .@Reset[5]) {
			mes "I'm sorry, but you have reached the Reset Limit.";	
			close;
		}
		if(.@Reset[5]){
			mes "You have "+callfunc("F_InsertPlural",.@Reset[5]-reset_limit,"Reset Limit")+" left. Do you want to proceed with the Stat Reset?";
			next;
			if(select("Cancel:Reset") == 1) close;
		}
		mes "[Hypnotist]";
		if (Zeny < .@Reset[1]) {
			mes "Sorry, you don't have enough Zeny.";
			close;
		}
		if (countitem(.@Reset[3]) < .@Reset[4]) {
			mes "Sorry, you don't have enough "+getitemname(.@Reset[3])+"s.";
			close;
		}
		set Zeny, Zeny-.@Reset[1];
		delitem .@Reset[3],.@Reset[4];
		ResetStatus;
		set reset_limit,reset_limit + 1;
		mes "All done!";
		close;
	case 3:
		mes "[Hypnotist]";
		mes "Do you want to proceed with the Skill Reset?";
		next;
		if(select("Cancel:Reset") == 1) close;
		mes "[Hypnotist]";
		if (Zeny < .@Reset[0]) {
			mes "Sorry, you don't have enough Zeny.";
			close;
		}
		set Zeny, Zeny-.@Reset[0];
		sc_end_class;
		ResetSkill;
		mes "All done!";
		close;
	case 4:
		mes "[Hypnotist]";
		mes "Okay, see you around.";
		close;
	}

OnInit:
	setarray .stat, bStr, bAgi, bVit, bInt, bDex, bLuk;
	end;
}
